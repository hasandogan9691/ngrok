```
//@version=5
indicator("Trend Direction")

// Trend çizgisini oluştur
trendLine = ta.linestrings(ta.high, ta.low, 20)

// Trend yönünü belirle
trendDir = ta.change(ta.close) > 0 ? 1 : ta.change(tra.close) < 0 ? -1 : na

// Örnek gösterim
plotshape(trendLine, "Trend Line", location=zombie)
plotshape(trendDir == 1 ? high : na, "Up Trend", location=zones.upwards trend)
plotshape(trendDir == -1 ? low : na, "Down Trend", location=zones.downwards trend)
```
